## What the module is for
Mission Control (a Textual dashboard) will show a "Local LLM" panel: is the Ollama model server running,
which models are installed, which model is loaded in GPU memory, and how full the GPU is. It can also
start/stop the Ollama user service. Your module provides the data; the UI is written by someone else
against EXACTLY the names below, so keep every name and signature.

## Exact API

OLLAMA_URL = "http://127.0.0.1:11434"
    Default server address. The environment variable MISSION_CONTROL_OLLAMA_URL overrides it.

@dataclass(frozen=True)
class InstalledModel:
    name: str
    size_bytes: int
    parameter_size: str | None      # from details.parameter_size, e.g. "9.0B"
    quantization: str | None        # from details.quantization_level, e.g. "Q4_K_M"

@dataclass(frozen=True)
class LoadedModel:
    name: str
    size_bytes: int                 # "size"
    vram_bytes: int                 # "size_vram"
    context_length: int | None      # "context_length"
    expires_at: datetime | None     # "expires_at", timezone-aware
    @property
    def gpu_percent(self) -> int: ...   # round(100 * vram_bytes / size_bytes); 0 if size_bytes is 0

@dataclass(frozen=True)
class GpuInfo:
    name: str
    memory_used_mb: int
    memory_total_mb: int
    utilization_percent: int | None    # None when nvidia-smi prints "[N/A]"

@dataclass(frozen=True)
class OllamaStatus:
    running: bool
    version: str | None = None
    installed: tuple[InstalledModel, ...] = ()
    loaded: tuple[LoadedModel, ...] = ()
    error: str | None = None

def parse_timestamp(value: object) -> datetime | None
    Ollama writes NANOseconds ("2026-11-18T13:11:50.080476753+02:00"). Cut the fraction to 6 digits before
    datetime.fromisoformat. Also accept a trailing "Z". Return None for anything that isn't a valid string.

def parse_tags(text: str) -> tuple[InstalledModel, ...]
    Parse the JSON reply of GET /api/tags. Raise ValueError for invalid JSON or when "models" isn't a list.
    Skip entries that aren't objects or have no string "name". Missing size -> 0; missing details -> None
    fields. Sort by name.

def parse_ps(text: str) -> tuple[LoadedModel, ...]
    Parse the JSON reply of GET /api/ps, same error rules as parse_tags. Missing numbers -> 0 / None.

def parse_nvidia_smi(text: str) -> GpuInfo | None
    Parse the FIRST line of `nvidia-smi --query-gpu=name,memory.used,memory.total,utilization.gpu
    --format=csv,noheader,nounits`, e.g. "NVIDIA GeForce RTX 5070, 9351, 12227, 18".
    Return None for empty or malformed output (never raise).

def host_command(argv: list[str]) -> list[str]
    Mission Control can run inside the distrobox container "dev", where systemctl --user and nvidia-smi
    belong to the HOST. If mission_control.containers.current_container_name() is not None, return
    ["distrobox-host-exec", *argv]; otherwise return argv unchanged.

def read_ollama(base_url: str | None = None, timeout: float = 1.5) -> OllamaStatus
    GET {url}/api/version with urllib.request. If it can't connect (URLError, OSError, timeout):
    return OllamaStatus(running=False) with NO error (a stopped server is normal, not an error).
    Otherwise GET /api/tags and /api/ps. If one of those fails or doesn't parse, still return
    running=True and the version, with error set to a short readable message.

def read_gpu(timeout: float = 3.0) -> GpuInfo | None
    Run host_command(["nvidia-smi", "--query-gpu=name,memory.used,memory.total,utilization.gpu",
    "--format=csv,noheader,nounits"]) with subprocess.run(capture_output=True, text=True, timeout=timeout,
    check=False). Return None on OSError, TimeoutExpired or a non-zero exit code.

def set_service(start: bool, timeout: float = 20.0) -> tuple[bool, str]
    Run host_command(["systemctl", "--user", "start" if start else "stop", "ollama"]).
    Success: (True, "Ollama started") or (True, "Ollama stopped").
    Failure: (False, "Could not start Ollama: <last line of stderr>") (or "stop"); also for OSError and
    TimeoutExpired, using the exception text.
